SELECT r.resulted_at, p.mrn, p.last_name, t.test_name, r.result_value, d.last_name AS doctor
FROM lab_result r
JOIN lab_order lo  ON lo.lab_order_id = r.lab_order_id
JOIN appointment a ON a.appointment_id = lo.appointment_id
JOIN patient p     ON p.patient_id = a.patient_id
JOIN doctor d      ON d.doctor_id = a.doctor_id
JOIN lab_test t    ON t.test_code = r.test_code
WHERE r.abnormal_flag = 'C' AND r.resulted_at >= TIMESTAMPTZ '2026-10-28' - interval '7 days'
ORDER BY r.resulted_at DESC;

SELECT a.scheduled_start, p.mrn, p.first_name, p.last_name, a.status, a.reason
FROM appointment a JOIN patient p USING (patient_id)
WHERE a.doctor_id = 15 AND a.scheduled_start >= TIMESTAMPTZ '2026-10-20' AND a.scheduled_start < TIMESTAMPTZ '2026-10-20' + interval '1 day'
ORDER BY a.scheduled_start;
